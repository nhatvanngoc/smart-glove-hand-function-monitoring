\begin{tikzpicture}[

  font=\sffamily,
  % Modern block style
  block/.style 2 args={
    draw=#1!90!black,
    line width=1.4pt,
    fill=#1!6,
    rounded corners=6pt,
    text width=#2,
    align=center,
    inner sep=8pt
  },
  % Arrows
  arrow/.style={
    ->,
    line width=1.8pt,

  },
  biarrow/.style={
    <->,
    line width=1.8pt,
    draw=#1!90!black,
    {Stealth[length=3.6mm, width=2.4mm]}-{Stealth[length=3.6mm, width=2.4mm]}
  },
  fbarrow/.style={
    ->,
    line width=1.4pt,
    dashed,
    draw=#1!85!black,
    -{Stealth[length=3.6mm, width=2.4mm]}
  },
  % Label on arrow
  lbl/.style={
    font=\small\bfseries,
    fill=white,
    inner sep=3.5pt,
    rounded corners=3pt,
    draw=gray!35,
    line width=0.7pt,
    text=#1!95!black
  }
]
